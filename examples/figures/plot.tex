% Built to plot.pdf, which examples/showcase.tex includes as a PDF figure.
\documentclass[border=4pt]{standalone}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\begin{document}
\begin{tikzpicture}
  \begin{axis}[width=9cm, height=5.5cm, xlabel={$x$}, ylabel={$y$},
      domain=0:6.3, samples=100, grid=major, legend pos=south west]
    \addplot[blue, thick] {sin(deg(x))};
    \addplot[red!80!black, thick, dashed] {cos(deg(x))};
    \legend{$\sin x$, $\cos x$}
  \end{axis}
\end{tikzpicture}
\end{document}
